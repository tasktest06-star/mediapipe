\section{Introduction}

The MediaPipe Hand Landmark model detects 21 3D keypoints of a hand skeleton from a single
RGB image at 30+~FPS on a CPU. This tutorial builds four applications: landmark overlay,
finger counting, rock/paper/scissors recognition, and gesture mouse control.

\section{Minimal Hand Detection}

\begin{lstlisting}[caption={Hand landmark overlay}]
import cv2, mediapipe as mp
mp_hands = mp.solutions.hands
mp_draw  = mp.solutions.drawing_utils

cap = cv2.VideoCapture(0)
with mp_hands.Hands(max_num_hands=2, model_complexity=0,
                    min_detection_confidence=0.5,
                    min_tracking_confidence=0.5) as hands:
    while cap.isOpened():
        ret, frame = cap.read()
        if not ret: break
        frame = cv2.flip(frame, 1)
        rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
        res = hands.process(rgb)
        if res.multi_hand_landmarks:
            for lms in res.multi_hand_landmarks:
                mp_draw.draw_landmarks(frame, lms,
                                       mp_hands.HAND_CONNECTIONS)
        cv2.imshow("Hands", frame)
        if cv2.waitKey(1) & 0xFF == ord('q'): break
cap.release(); cv2.destroyAllWindows()
\end{lstlisting}

\section{Finger Counting}

A finger is up when its tip $y$-coordinate is \emph{less than} its MCP $y$-coordinate
(smaller $y$ = higher on screen). The thumb uses $x$-coordinates due to its lateral motion.

\begin{lstlisting}[caption={Finger counting logic}]
FINGER_TIPS = [4, 8, 12, 16, 20]
FINGER_MCPS = [2, 5,  9, 13, 17]

def count_fingers(lms, label):
    lm = lms.landmark
    up = 0
    # Thumb
    if label == "Right":
        up += 1 if lm[4].x < lm[2].x else 0
    else:
        up += 1 if lm[4].x > lm[2].x else 0
    # Fingers 2-5
    for tip, mcp in zip(FINGER_TIPS[1:], FINGER_MCPS[1:]):
        up += 1 if lm[tip].y < lm[mcp].y else 0
    return up
\end{lstlisting}

\section{Rock / Paper / Scissors}

\begin{center}
\begin{tabular}{lcc}
\toprule
\textbf{Gesture} & \textbf{Fingers up} & \textbf{Classification rule} \\
\midrule
Rock     & 0 & All fingers down \\
Paper    & 5 & All fingers up \\
Scissors & 2 & Index + middle up only \\
\bottomrule
\end{tabular}
\end{center}

\begin{lstlisting}[caption={Gesture classifier}]
def classify_gesture(state):  # state = [thumb, idx, mid, ring, pinky]
    if sum(state) == 0:   return "Rock"
    if sum(state) == 5:   return "Paper"
    if state[1] and state[2] and not state[3] and not state[4]:
        return "Scissors"
    return "Unknown"
\end{lstlisting}
